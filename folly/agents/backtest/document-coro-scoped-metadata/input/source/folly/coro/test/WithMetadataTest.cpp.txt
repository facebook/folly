using namespace folly;
using namespace folly::coro;

namespace {
FOLLY_NOINLINE std::vector<std::uint64_t> getMetadata() {
  constexpr size_t kMaxSize = 100;
  uintptr_t addrArr[kMaxSize] = {};
  std::uint64_t metadataArr[kMaxSize] = {};
  auto size =
      symbolizer::getAsyncStackTraceSafe(addrArr, metadataArr, kMaxSize);

  std::vector<std::uint64_t> res;
  for (ssize_t i = 0; i < size; ++i) {
    if (metadataArr[i] != 0) {
      res.push_back(metadataArr[i]);
    }
  }
  return res;
}

template <typename... Args>
std::vector<std::uint64_t> makeMetadata(Args&&... args) {
  return {static_cast<std::uint64_t>(std::forward<Args>(args))...};
}
} // namespace

TEST(WithMetadata, OneExecutorNoMetadata) {
  CPUThreadPoolExecutor exec{1};

  std::vector<std::uint64_t> metadata;

  auto func3 = [&]() -> Task<void> {
    metadata = getMetadata();
    co_return;
  };
  auto func2 = [&]() -> Task<void> { co_await func3(); };
  auto func1 = [&]() -> Task<void> { co_await func2(); };

  blockingWait(co_withExecutor(&exec, func1()));

  EXPECT_EQ(metadata, makeMetadata());
}

TEST(WithMetadata, OneExecutorOneMetadata) {
